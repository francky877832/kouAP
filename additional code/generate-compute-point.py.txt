from graphviz import Digraph

# Création du graphe
dot = Digraph(comment="Flowchart of computes_points")

# Définition des nœuds (étapes)
dot.node('A', 'Start')
dot.node('B', 'Loop: for i in forms_datas')
dot.node('C', 'Check: len(act) <= 0')
dot.node('D', 'Continue')
dot.node('E', 'Set letter = chr(65 + i)')
dot.node('F', 'Find activity by letter')
dot.node('G', 'Loop: for j in act')
dot.node('H', 'Set number, normal_point, activity_points')
dot.node('I', 'Check: case_arr.length > 0 and "no_case" not in case_arr')
dot.node('J', 'Get case and participant_info')
dot.node('K', 'activity_points = normal_point * coef * coef2')
dot.node('L', 'Check: letter in cased_activities')
dot.node('M', 'Get coef and factor by num_writer')
dot.node('N', 'activity_points = normal_point * (coef / factor)')
dot.node('O', 'activity_points = normal_point')
dot.node('P', 'update_points(letter, number, round(activity_points))')
dot.node('Z', 'End')

# Connexions entre les nœuds
dot.edge('A', 'B')
dot.edge('B', 'C', label='for each i')
dot.edge('C', 'D', label='True')
dot.edge('C', 'E', label='False')
dot.edge('E', 'F')
dot.edge('F', 'G')
dot.edge('G', 'H', label='for each j')
dot.edge('H', 'I')
dot.edge('I', 'J', label='True')
dot.edge('J', 'K')
dot.edge('K', 'P')
dot.edge('I', 'L', label='False')
dot.edge('L', 'M', label='True')
dot.edge('M', 'N')
dot.edge('N', 'P')
dot.edge('L', 'O', label='False')
dot.edge('O', 'P')
dot.edge('P', 'G', label='Next j')
dot.edge('G', 'B', label='Next i')
dot.edge('D', 'B')
dot.edge('B', 'Z', label='End loop')

# Sauvegarde en image
dot.render('computes_points_flowchart', format='png', cleanup=False)
