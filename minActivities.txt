import React, { useContext } from "react";
import { UserContext } from "../context/UserContext";
import "bootstrap/dist/css/bootstrap.min.css";

const MinActivities = () => {
  const { minActivities } = useContext(UserContext);

  // Récupérer toutes les facultés uniques
  const faculties = Array.from(
    new Set(minActivities.flatMap((activity) => activity.positions.map((pos) => pos.faculty.name)))
  );

  return (
    <div className="container mt-4">
      <div className="table-responsive">
        <table className="table table-bordered">
          <thead className="table-dark">
            <tr>
              <th className="text-center">Activité</th>
              {faculties.map((faculty) => (
                <th key={faculty} colSpan={3} className="text-center">{faculty}</th>
              ))}
            </tr>
            <tr className="table-secondary">
              <th></th>
              {faculties.map(() =>
                ["Assistant Prof", "Associate Prof", "Professor"].map((title) => (
                  <th key={title} className="text-center">{title}</th>
                ))
              )}
            </tr>
          </thead>
          <tbody>
            {minActivities.map((activity, index) => (
              <tr key={index}>
                {/* Colonne Activité */}
                <td className="text-center fw-bold">
                  {activity.range
                    ? `${activity.letter}${activity.from} - ${activity.letter}${activity.to}`
                    : activity.criteria}
                </td>

                {/* Colonnes des Facultés */}
                {faculties.map((faculty) => {
                  // Trouver les positions correspondantes
                  const facultyPositions = activity.positions.filter(
                    (pos) => pos.faculty.name === faculty
                  );

                  return [0, 1, 2].map((position) => {
                    const foundPosition = facultyPositions.find((pos) => pos.position === position);
                    return (
                      <td key={`${faculty}-${position}`} className="text-center">
                        {foundPosition ? foundPosition.quantity : "-"}
                      </td>
                    );
                  });
                })}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
};

export default MinActivities;
